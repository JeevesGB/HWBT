struct VS_INPUT
{
	float4 Pos 		: POSITION;
	float4 Diffuse 	: COLOR0;
	float3 Normal 	: NORMAL;
	float3 Tangent 	: TEXCOORD0;
};

struct VS_OUTPUT
{
	float4 Pos 		 	: POSITION;
	float4 Diffuse 		: COLOR0;
	float3 LookupDir 	: TEXCOORD0;
	float3 LookupDir1	: TEXCOORD1;
	float3 LookupDir2	: TEXCOORD2;
	float3 LookupDir3	: TEXCOORD3;
	float3 LookupDir4	: TEXCOORD4;
	float3 LookupDir5	: TEXCOORD5;
	float3 LookupDir6	: TEXCOORD6;
};

int sys_NumBlurPasses = 3;

float4x4 g_WorldViewProjection : WORLDVIEWPROJECTION;
float g_SourceLevel;	// miplevel to sample from
float g_DestDims;		// size of current face
float g_SeamFixup;		// seam fixup, push it in by n * Normal
int g_SourcePO2;		// source size (^2)

samplerCUBE g_Tex0 : register(s0) = sampler_state
{
	MinFilter = Linear;
	MagFilter = Linear;
	MipFilter = Point;
	AddressU = Wrap;
	AddressV = Wrap;
	AddressW = Wrap;
};

float4x4 MatrixRotX(float ang)
{
	float c; float s;
	sincos(ang, s, c);
	
	float4x4 m;
	m[0][0] = 1; m[0][1] = 0; m[0][2] =  0; m[0][3] = 0;
	m[1][0] = 0; m[1][1] = c; m[1][2] = -s; m[1][3] = 0; 
	m[2][0] = 0; m[2][1] = s; m[2][2] =  c; m[2][3] = 0;
	m[3][0] = 0; m[3][1] = 0; m[3][2] =  0; m[3][3] = 1;
	
	return m;
}

float4x4 MatrixRotY(float ang)
{
	float c; float s;
	sincos(ang, s, c);
	
	float4x4 m;
	m[0][0] = c; m[0][1] = 0; m[0][2] = -s; m[0][3] = 0;
	m[1][0] = 0; m[1][1] = 1; m[1][2] =  0; m[1][3] = 0; 
	m[2][0] = s; m[2][1] = 0; m[2][2] =  c; m[2][3] = 0;
	m[3][0] = 0; m[3][1] = 0; m[3][2] =  0; m[3][3] = 1;
	
	return m;
}

float4x4 MatrixRotZ(float ang)
{
	float c; float s;
	sincos(ang, s, c);
	
	float4x4 m;
	m[0][0] = c; m[0][1] =-s; m[0][2] =  0; m[0][3] = 0;
	m[1][0] = s; m[1][1] = c; m[1][2] =  0; m[1][3] = 0; 
	m[2][0] = 0; m[2][1] = 0; m[2][2] =  1; m[2][3] = 0;
	m[3][0] = 0; m[3][1] = 0; m[3][2] =  0; m[3][3] = 1;
	
	return m;
}

VS_OUTPUT vs_y(VS_INPUT In)
{
	VS_OUTPUT Out = (VS_OUTPUT)0;
	float4 pos = In.Pos;
	pos.z *= -1;
	
	float tmp_SeamFixup = blurVals[g_SourcePO2];
	
	Out.Pos = mul(pos, g_WorldViewProjection);
	//Out.LookupDir = lerp(normalize(In.Pos), In.Normal, g_SeamFixup * SeamScale);
	Out.LookupDir = lerp(normalize(In.Pos), In.Normal, tmp_SeamFixup);

	Out.LookupDir1 = mul(Out.LookupDir, MatrixRotY(0.01745  * blurMults[0])).xyz;
	Out.LookupDir2 = mul(Out.LookupDir, MatrixRotY(-0.01745 * blurMults[0])).xyz;
	Out.LookupDir3 = mul(Out.LookupDir, MatrixRotY(0.01745  * blurMults[1])).xyz;
	Out.LookupDir4 = mul(Out.LookupDir, MatrixRotY(-0.01745 * blurMults[1])).xyz;
	Out.LookupDir5 = mul(Out.LookupDir, MatrixRotY(0.01745  * blurMults[2])).xyz;
	Out.LookupDir6 = mul(Out.LookupDir, MatrixRotY(-0.01745 * blurMults[2])).xyz;

	Out.Diffuse = In.Diffuse;
	
	return Out;
}
